// AI Harness — the brain behind ilpro.ai interface HarnessConfig { models: ModelConfig[]; fallbackChain: string[]; maxRetries: 3; timeout: 10000; } interface ModelConfig { name: string; endpoint: string; costPerReq: number; tasks: string[]; // which tasks this model handles } async function runHarness(input: UserInput): Promise<AIResponse> { // 1. CLASSIFY — what kind of task is this? const task = classifyTask(input.text); // → "review_reply" | "website_gen" | "chatbot" | "quote" | "marketing" // 2. ROUTE — pick the best model for this task const model = selectModel(task, input.plan); // → Free plan: Qwen local → Pro plan: Claude Haiku/Sonnet // 3. BUILD PROMPT — inject context into template const prompt = buildPrompt(task, { business: input.businessInfo, context: input.text, outputFormat: "json", }); // 4. CALL MODEL — with fallback chain for (const attempt of config.fallbackChain) { try { const raw = await callModel(attempt, prompt); // 5. PARSE — validate and extract structured data const parsed = parseOutput(raw, task.schema); if (!parsed.valid) throw new Error("Invalid output"); // 6. GUARDRAILS — safety checks const safe = checkGuardrails(parsed.data); if (!safe) throw new Error("Safety violation"); // 7. LOG — track cost, latency, model used logUsage({ model: attempt, cost: model.costPerReq, latency }); return parsed.data; } catch (e) { logError(attempt, e); continue; // try next model in fallback chain } } // All models failed — return template response return getTemplateResponse(task); }
